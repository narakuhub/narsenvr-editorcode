local function ResolveIcon(iconStr)
    if not iconStr or iconStr == "" then return "" end
    if iconStr:find("rbxassetid://") or iconStr:find("http") then
        return iconStr
    end
    return IconMap[iconStr] or ""
end

-- Drag System
local dragging = false
local dragInput, dragStart, startPos

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Window.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Resize System
local function SetupResize(handle, resizeX, resizeY)
    local resizing = false
    local resizeStart, startSize

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            resizing = true
            resizeStart = input.Position
            startSize = Window.Size
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    resizing = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if resizing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - resizeStart
            local newX = startSize.X.Offset
            local newY = startSize.Y.Offset

            if resizeX then
                newX = math.clamp(startSize.X.Offset + delta.X, 220, 800)
            end
            if resizeY then
                newY = math.clamp(startSize.Y.Offset + delta.Y, 200, 600)
            end

            Window.Size = UDim2.new(0, newX, 0, newY)
        end
    end)
end

SetupResize(BottomResizeHandle, false, true)
SetupResize(RightResizeHandle, true, false)
SetupResize(BottomRightResizeHandle, true, true)

        TweenService:Create(Window, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

    end
end)

    ScreenGui:Destroy()
end)

local ComponentRegistry = {}
local ActiveTab = nil

local WindUI = {}

function WindUI:CreateWindow(config)
    config = config or {}
    Title.Text = config.Title or "WindUI"
    Subtitle.Text = config.Subtitle or "v1.0.0"
    if config.Size then
        Window.Size = config.Size
    end
    if config.Icon then
        Logo.Image = ResolveIcon(config.Icon)
    end

    local WindowObj = {}

    function WindowObj:CreateTab(tabConfig)
        tabConfig = tabConfig or {}
        local tabNameStr = tabConfig.Name or "Tab"
        local tabIconStr = ResolveIcon(tabConfig.Icon)

        -- Build Tab Button
        local TabBtn = TabTemplate:Clone()
        TabBtn.Name:GetParent().Name.Text = tabNameStr
        TabBtn.Icon.Image = tabIconStr
        TabBtn.Visible = true
        TabBtn.Parent = TabBar

        -- Build Page Container
        local Page = Instance.new("ScrollingFrame")
        Page.Name = tabNameStr .. "_Page"
        Page.Size = UDim2.new(1, 0, 1, 0)
        Page.BackgroundTransparency = 1
        Page.CanvasSize = UDim2.new(0, 0, 0, 0)
        Page.AutomaticCanvasSize = Enum.AutomaticSize.Y
        Page.ScrollBarThickness = 3
        Page.ScrollBarImageColor3 = Theme.Border
        Page.Visible = false
        Page.Parent = Pages

        local PageList = Instance.new("UIListLayout")
        PageList.SortOrder = Enum.SortOrder.LayoutOrder
        PageList.Padding = UDim.new(0, 6)
        PageList.Parent = Page

        local PagePadding = Instance.new("UIPadding")
        PagePadding.PaddingLeft = UDim.new(0, 8)
        PagePadding.PaddingRight = UDim.new(0, 8)
        PagePadding.PaddingTop = UDim.new(0, 6)
        PagePadding.PaddingBottom = UDim.new(0, 6)
        PagePadding.Parent = Page

        -- Tab Switching
        local function SelectTab()
            for _, child in ipairs(Pages:GetChildren()) do
                if child:IsA("ScrollingFrame") then
                    child.Visible = false
                end
            end
            for _, child in ipairs(TabBar:GetChildren()) do
                if child:IsA("TextButton") then
                    TweenService:Create(child.Name, TweenInfo.new(0.2), {TextColor3 = Theme.TextSecondary}):Play()
                    TweenService:Create(child.Icon, TweenInfo.new(0.2), {ImageColor3 = Theme.TextSecondary}):Play()
                    child.BackgroundTransparency = 1
                end
            end

            Page.Visible = true
            TabBtn.BackgroundTransparency = 0.5
            TweenService:Create(TabBtn.Name, TweenInfo.new(0.2), {TextColor3 = Theme.Accent}):Play()
            TweenService:Create(TabBtn.Icon, TweenInfo.new(0.2), {ImageColor3 = Theme.Accent}):Play()
        end

        TabBtn.MouseButton1Click:Connect(SelectTab)

        if ActiveTab == nil then
            ActiveTab = TabBtn
            SelectTab()
        end

        local TabObj = {}

        -- COMPONENT: Section
        function TabObj:CreateSection(secConfig)
            secConfig = secConfig or {}
            local sec = SectionTemplate:Clone()
            sec.Name.Text = (secConfig.Name or "SECTION"):upper()
            sec.Icon.Image = ResolveIcon(secConfig.Icon)
            sec.Visible = true
            sec.Parent = Page
            return sec
        end

        -- COMPONENT: Toggle
        function TabObj:CreateToggle(toggleConfig)
            toggleConfig = toggleConfig or {}
            local toggleInstance = ToggleTemplate:Clone()
            local textContainer = toggleInstance.TextContainer
            local track = toggleInstance.ToggleTrack
            local knob = track.ToggleKnob

            toggleInstance.Icon.Image = ResolveIcon(toggleConfig.Icon)
            textContainer.Name.Text = toggleConfig.Name or "Toggle"
            textContainer.Description.Text = toggleConfig.Description or ""
            toggleInstance.Visible = true
            toggleInstance.Parent = Page

            local state = toggleConfig.Default or false

            local function UpdateState(val)
                state = val
                if state then
                    TweenService:Create(track, TweenInfo.new(0.2), {BackgroundColor3 = Theme.Accent}):Play()
                    TweenService:Create(knob, TweenInfo.new(0.2), {Position = UDim2.new(1, -18, 0.5, -8)}):Play()
                else
                    TweenService:Create(track, TweenInfo.new(0.2), {BackgroundColor3 = Theme.ToggleOffTrack}):Play()
                    TweenService:Create(knob, TweenInfo.new(0.2), {Position = UDim2.new(0, 2, 0.5, -8)}):Play()
                end
                if toggleConfig.Callback then
                    toggleConfig.Callback(state)
                end
            end

            UpdateState(state)

            track.MouseButton1Click:Connect(function()
                UpdateState(not state)
            end)

            local ToggleObj = {}
            function ToggleObj:Set(val) UpdateState(val) end
            function ToggleObj:Get() return state end
            function ToggleObj:Destroy() toggleInstance:Destroy() end
            return ToggleObj
        end

        -- COMPONENT: Slider
        function TabObj:CreateSlider(sliderConfig)
            sliderConfig = sliderConfig or {}
            local sliderInstance = SliderTemplate:Clone()
            local textContainer = sliderInstance.TextContainer
            local track = sliderInstance.SliderTrack
            local fill = track.Fill

            sliderInstance.Icon.Image = ResolveIcon(sliderConfig.Icon)
            textContainer.Name.Text = sliderConfig.Name or "Slider"
            sliderInstance.Visible = true
            sliderInstance.Parent = Page

            local min = sliderConfig.Min or 0
            local max = sliderConfig.Max or 100
            local step = sliderConfig.Step or 1
            local value = sliderConfig.Default or min

            local function Snap(val)
                return math.floor((val - min) / step + 0.5) * step + min
            end

            local function UpdateSlider(val)
                value = math.clamp(Snap(val), min, max)
                textContainer.Value.Text = tostring(value)
                local percent = (value - min) / (max - min)
                fill.Size = UDim2.new(percent, 0, 1, 0)
                if sliderConfig.Callback then
                    sliderConfig.Callback(value)
                end
            end

            UpdateSlider(value)

            local sliding = false

            track.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = true
                    local posX = input.Position.X - track.AbsolutePosition.X
                    local pct = math.clamp(posX / track.AbsoluteSize.X, 0, 1)
                    UpdateSlider(min + (max - min) * pct)
                end
            end)

            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = false
                end
            end)

            UserInputService.InputChanged:Connect(function(input)
                if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    local posX = input.Position.X - track.AbsolutePosition.X
                    local pct = math.clamp(posX / track.AbsoluteSize.X, 0, 1)
                    UpdateSlider(min + (max - min) * pct)
                end
            end)

            local SliderObj = {}
            function SliderObj:Set(val) UpdateSlider(val) end
            function SliderObj:Get() return value end
            function SliderObj:Destroy() sliderInstance:Destroy() end
            return SliderObj
        end

        -- COMPONENT: Dropdown
        function TabObj:CreateDropdown(dropConfig)
            dropConfig = dropConfig or {}
            local dropInstance = DropdownTemplate:Clone()
            local textContainer = dropInstance.TextContainer
            local optionsContainer = dropInstance.OptionsContainer

            dropInstance.Icon.Image = ResolveIcon(dropConfig.Icon)
            textContainer.Name.Text = dropConfig.Name or "Dropdown"
            dropInstance.Visible = true
            dropInstance.Parent = Page

            local isOpen = false
            local selected = dropConfig.Default or nil
            local options = dropConfig.Options or {}

            local function ToggleOpen()
                isOpen = not isOpen
                local count = #options
                local targetHeight = isOpen and math.min(count * 30, 120) or 0
                local totalHeight = 42 + targetHeight

                TweenService:Create(dropInstance, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, totalHeight)
                }):Play()

                TweenService:Create(optionsContainer, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, targetHeight)
                }):Play()

                TweenService:Create(dropInstance.Arrow, TweenInfo.new(0.2), {
                    Rotation = isOpen and 180 or 0
                }):Play()
            end

            local function RefreshOptions()
                for _, child in ipairs(optionsContainer:GetChildren()) do
                    if child:IsA("TextButton") then
                        child:Destroy()
                    end
                end

                for _, opt in ipairs(options) do
                    local card = OptionCardTemplate:Clone()
                    card.Name.Text = tostring(opt)
                    card.SelectedIndicator.Visible = (opt == selected)
                    card.Visible = true
                    card.Parent = optionsContainer

                    card.MouseButton1Click:Connect(function()
                        selected = opt
                        textContainer.SelectedValue.Text = tostring(opt)
                        RefreshOptions()
                        ToggleOpen()
                        if dropConfig.Callback then
                            dropConfig.Callback(selected)
                        end
                    end)
                end
                optionsContainer.CanvasSize = UDim2.new(0, 0, 0, #options * 30)
            end

            if selected then
                textContainer.SelectedValue.Text = tostring(selected)
            end

            RefreshOptions()

            dropInstance.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    if input.Position.Y - dropInstance.AbsolutePosition.Y <= 42 then
                        ToggleOpen()
                    end
                end
            end)

            local DropdownObj = {}
            function DropdownObj:Set(val)
                selected = val
                textContainer.SelectedValue.Text = tostring(val)
                RefreshOptions()
            end
            function DropdownObj:SetOptions(newOpts)
                options = newOpts
                RefreshOptions()
            end
            function DropdownObj:Get() return selected end
            function DropdownObj:Destroy() dropInstance:Destroy() end
            return DropdownObj
        end

        -- COMPONENT: Button
        function TabObj:CreateButton(btnConfig)
            btnConfig = btnConfig or {}
            local btnInstance = ButtonTemplate:Clone()
            local trigger = btnInstance.Trigger
            local textContainer = trigger.TextContainer

            trigger.Icon.Image = ResolveIcon(btnConfig.Icon)
            textContainer.Name.Text = btnConfig.Name or "Button"
            textContainer.Description.Text = btnConfig.Description or ""
            btnInstance.Visible = true
            btnInstance.Parent = Page

            trigger.MouseButton1Click:Connect(function()
                TweenService:Create(btnInstance, TweenInfo.new(0.1), {BackgroundTransparency = 0.1}):Play()
                task.delay(0.1, function()
                    TweenService:Create(btnInstance, TweenInfo.new(0.1), {BackgroundTransparency = Theme.CardTrans}):Play()
                end)
                if btnConfig.Callback then
                    btnConfig.Callback()
                end
            end)

            local ButtonObj = {}
            function ButtonObj:Destroy() btnInstance:Destroy() end
            return ButtonObj
        end

        -- COMPONENT: Input
        function TabObj:CreateInput(inputConfig)
            inputConfig = inputConfig or {}
            local inputInstance = InputTemplate:Clone()
            local textBox = inputInstance.TextBox
            local clearBtn = inputInstance.ClearButton

            inputInstance.Icon.Image = ResolveIcon(inputConfig.Icon)
            inputInstance.Name.Text = inputConfig.Name or "Input"
            textBox.PlaceholderText = inputConfig.Placeholder or "Enter text..."
            textBox.Text = inputConfig.Default or ""
            inputInstance.Visible = true
            inputInstance.Parent = Page

            clearBtn.Visible = (textBox.Text ~= "")

            textBox:GetPropertyChangedSignal("Text"):Connect(function()
                clearBtn.Visible = (textBox.Text ~= "")
            end)

            clearBtn.MouseButton1Click:Connect(function()
                textBox.Text = ""
            end)

            textBox.FocusLost:Connect(function(enterPressed)
                if inputConfig.Callback then
                    inputConfig.Callback(textBox.Text, enterPressed)
                end
            end)

            local InputObj = {}
            function InputObj:Set(val) textBox.Text = val end
            function InputObj:Get() return textBox.Text end
            function InputObj:Destroy() inputInstance:Destroy() end
            return InputObj
        end

        return TabObj
    end

    return WindowObj
end
