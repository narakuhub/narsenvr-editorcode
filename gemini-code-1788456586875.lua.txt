-- OpenButton Draggable implementation (doesn't trigger open action on drag)
local openDragging = false
local openDragStart, openStartPos
local wasDragged = false

OpenButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		openDragging = true
		wasDragged = false
		openDragStart = input.Position
		openStartPos = OpenButton.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				openDragging = false
			end
		end)
	end
end)

Services.UserInputService.InputChanged:Connect(function(input)
	if openDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - openDragStart
		if delta.Magnitude > 5 then
			wasDragged = true
			OpenButton.Position = UDim2.new(openStartPos.X.Scale, openStartPos.X.Offset + delta.X, openStartPos.Y.Scale, openStartPos.Y.Offset + delta.Y)
		end
	end
end)

	local timeStr = os.date("%H:%M:%S")
	local formattedText = string.format("[%s] %s", timeStr, text)
	table.insert(consoleLogs, formattedText)
	if #consoleLogs > 200 then
		table.remove(consoleLogs, 1)
	end

	ConsoleScroll.CanvasPosition = Vector2.new(0, math.max(0, ConsoleScroll.CanvasSize.Y.Offset - ConsoleScroll.AbsoluteWindowSize.Y))
end

local function setupResizeHandle(handleFrame, resizeX, resizeY)
	local resizing = false
	local startSize, startMousePos

	handleFrame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			resizing = true
			startSize = MainFrame.Size
			startMousePos = input.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then 
					resizing = false 
				end
			end)
		end
	end)

	Services.UserInputService.InputChanged:Connect(function(input)
		if resizing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - startMousePos
			local newW = resizeX and math.max(450, startSize.X.Offset + delta.X) or MainFrame.Size.X.Offset
			local newH = resizeY and math.clamp(startSize.Y.Offset + delta.Y, 320, 600) or MainFrame.Size.Y.Offset
			MainFrame.Size = UDim2.fromOffset(newW, newH)
			lastCustomSize = MainFrame.Size
		end
	end)
end

local function updateCustomScrollbars()
	local windowY = CodeScroll.AbsoluteWindowSize.Y
	local canvasY = CodeScroll.CanvasSize.Y.Offset
	if canvasY > windowY and canvasY > 0 then
		local thumbRatioY = math.clamp(windowY / canvasY, 0.1, 1)
		local trackHeight = VScrollTrack.AbsoluteSize.Y - 28
		local thumbHeight = math.max(16, trackHeight * thumbRatioY)
		VScrollThumb.Size = UDim2.new(1, 0, 0, thumbHeight)

		local maxScrollY = canvasY - windowY
		local scrollRatioY = (maxScrollY > 0) and (CodeScroll.CanvasPosition.Y / maxScrollY) or 0
		local maxThumbY = trackHeight - thumbHeight
		VScrollThumb.Position = UDim2.new(0, 0, 0, 14 + (maxThumbY * math.clamp(scrollRatioY, 0, 1)))
		VScrollTrack.Visible = true
	else
		VScrollTrack.Visible = false
	end

	local windowX = CodeScroll.AbsoluteWindowSize.X
	local canvasX = CodeScroll.CanvasSize.X.Offset
	if canvasX > windowX and canvasX > 0 then
		local thumbRatioX = math.clamp(windowX / canvasX, 0.1, 1)
		local trackWidth = HScrollTrack.AbsoluteSize.X - 28
		local thumbWidth = math.max(16, trackWidth * thumbRatioX)
		HScrollThumb.Size = UDim2.new(0, thumbWidth, 1, 0)

		local maxScrollX = canvasX - windowX
		local scrollRatioX = (maxScrollX > 0) and (CodeScroll.CanvasPosition.X / maxScrollX) or 0
		local maxThumbX = trackWidth - thumbWidth
		HScrollThumb.Position = UDim2.new(0, 14 + (maxThumbX * math.clamp(scrollRatioX, 0, 1)), 0, 0)
		HScrollTrack.Visible = true
	else
		HScrollTrack.Visible = false
	end
end

			CodeScroll.CanvasPosition = Vector2.new(CodeScroll.CanvasPosition.X, math.clamp(vStartScroll + scrollDelta, 0, maxScroll))
		end
	end

local function escapeRichText(str)
	return str:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")
end

local function applyHighlighting(source)
	local tokens = {}
	local count = 0
	local i = 1
	local len = #source

	local function appendToken(val)
		count = count + 1
		tokens[count] = val
	end

	while i <= len do
		if source:sub(i, i+1) == "--" then
			local endPos = source:find("\n", i) or (len + 1)
			local commentText = source:sub(i, endPos - 1)
			appendToken(string.format('<font color="#6A9955">%s</font>', escapeRichText(commentText)))
			i = endPos
		elseif source:sub(i, i) == '"' or source:sub(i, i) == "'" then
			local quote = source:sub(i, i)
			local endPos = i + 1
			while endPos <= len do
				if source:sub(endPos, endPos) == quote and source:sub(endPos - 1, endPos - 1) ~= "\\" then
					break
				end
				endPos = endPos + 1
			end
			local strText = source:sub(i, math.min(endPos, len))
			appendToken(string.format('<font color="#CE9178">%s</font>', escapeRichText(strText)))
			i = endPos + 1
		elseif source:sub(i, i):match("[%a_]") then
			local startPos = i
			while i <= len and source:sub(i, i):match("[%w_]") do
				i = i + 1
			end
			local word = source:sub(startPos, i - 1)
			local prevChar = source:sub(startPos - 1, startPos - 1)

			local isKw = false
			for _, kw in ipairs(LUAU_KEYWORDS) do
				if kw == word then
					isKw = true
					break
				end
			end

			if isKw then
				appendToken(string.format('<font color="#C586C0">%s</font>', word))
			elseif prevChar == "." or prevChar == ":" then
				appendToken(string.format('<font color="#64B5F6">%s</font>', word))
			elseif API_DATABASE[word] then
				appendToken(string.format('<font color="#4EC9B0">%s</font>', word))
			else
				appendToken(escapeRichText(word))
			end
		elseif source:sub(i, i):match("%d") then
			local startPos = i
			while i <= len and source:sub(i, i):match("[%d%.]") do
				i = i + 1
			end
			local num = source:sub(startPos, i - 1)
			appendToken(string.format('<font color="#B5CEA8">%s</font>', num))
		else
			appendToken(escapeRichText(source:sub(i, i)))
			i = i + 1
		end
	end

	return table.concat(tokens)
end

local function parseUserSymbols(source)
	local symbols = {}
	local seen = {}
	local inferred = {}

	for line in source:gmatch("[^\r\n]+") do
		local svcVar, svcName = line:match("local%s+([%w_]+)%s*=%s*game:GetService%([\"'](%w+)[\"']%)")
		if svcVar and svcName then inferred[svcVar] = svcName end

		local instVar, instClass = line:match("local%s+([%w_]+)%s*=%s*Instance%.new%([\"'](%w+)[\"']%)")
		if instVar and instClass then inferred[instVar] = instClass end

		for fn, params in line:gmatch("function%s+([%w_]+)%s*%(([^%)]*)%)") do
			if not seen[fn] then
				seen[fn] = true
				table.insert(symbols, { name = fn .. "()", category = "FUNCTION", cleanName = fn })
			end
			for p in params:gmatch("[%w_]+") do
				if not seen[p] and not LUAU_KEYWORDS[p] then
					seen[p] = true
					table.insert(symbols, { name = p, category = "PARAMETER", cleanName = p })
				end
			end
		end

		for fn, params in line:gmatch("local%s+function%s+([%w_]+)%s*%(([^%)]*)%)") do
			if not seen[fn] then
				seen[fn] = true
				table.insert(symbols, { name = fn .. "()", category = "FUNCTION", cleanName = fn })
			end
			for p in params:gmatch("[%w_]+") do
				if not seen[p] and not LUAU_KEYWORDS[p] then
					seen[p] = true
					table.insert(symbols, { name = p, category = "PARAMETER", cleanName = p })
				end
			end
		end

		for vars in line:gmatch("for%s+([^in]+)%s+in") do
			for v in vars:gmatch("[%w_]+") do
				if not seen[v] and not LUAU_KEYWORDS[v] then
					seen[v] = true
					table.insert(symbols, { name = v, category = "LOOP VARIABLE", cleanName = v })
				end
			end
		end

		for v in line:gmatch("local%s+([%w_]+)") do
			if v ~= "function" and not seen[v] then
				seen[v] = true
				table.insert(symbols, { name = v, category = "VARIABLE", cleanName = v })
			end
		end
	end

	return symbols, inferred
end

local function getCursorDetails()
	local text = CodeInput.Text
	local cursorPos = math.max(1, CodeInput.CursorPosition)
	local subText = text:sub(1, cursorPos - 1)
	local lines = string.split(text, "\n")
	local subLines = string.split(subText, "\n")

	return #subLines, #subLines[#subLines] + 1, #lines, lines
end

	HighlightLabel.Size = UDim2.new(1, 0, 0, math.max(contentHeight, viewportHeight))

	updateCustomScrollbars()

local function getSuggestions(token, contextVar)
	local suggestions = {}
	local seen = {}

	local userSyms, inferredTypes = parseUserSymbols(CodeInput.Text)
	local targetContext = contextVar
	if contextVar and inferredTypes[contextVar] then
		targetContext = inferredTypes[contextVar]
	end

	if targetContext and API_DATABASE[targetContext] then
		for _, m in ipairs(API_DATABASE[targetContext]) do
			if m:lower():sub(1, #token) == token:lower() and not seen[m] then
				seen[m] = true
				table.insert(suggestions, { display = m, insert = m, category = "API" })
			end
		end
	elseif not contextVar then
		for _, sym in ipairs(userSyms) do
			if sym.cleanName:lower():sub(1, #token) == token:lower() and not seen[sym.name] then
				seen[sym.name] = true
				table.insert(suggestions, { display = sym.name, insert = sym.cleanName, category = sym.category })
			end
		end

		for k, _ in pairs(API_DATABASE) do
			if k:lower():sub(1, #token) == token:lower() and not seen[k] then
				seen[k] = true
				table.insert(suggestions, { display = k, insert = k, category = "API" })
			end
		end

		for _, kw in ipairs(LUAU_KEYWORDS) do
			if kw:lower():sub(1, #token) == token:lower() and not seen[kw] then
				seen[kw] = true
				table.insert(suggestions, { display = kw, insert = kw, category = "KEYWORD" })
			end
		end
	end

	return suggestions
end

local function acceptSuggestion()
	if not AutoPopup.Visible or #activeSuggestions == 0 then return end
	local selected = activeSuggestions[selectedIndex]
	if not selected then return end

	local text = CodeInput.Text
	local pos = CodeInput.CursorPosition
	local left = text:sub(1, pos - 1)
	local right = text:sub(pos)

	local contextVar, token = left:match("([%w_]*)[%.:]([%w_]*)$")
	if not token then token = left:match("([%w_]+)$") or "" end

	local newLeft = left:sub(1, #left - #token) .. selected.insert
	CodeInput.Text = newLeft .. right
	CodeInput.CursorPosition = #newLeft + 1
	AutoPopup.Visible = false
end

local parseDebounce = false
local function checkAutocomplete()
	if not isEditFocusMode or parseDebounce then return end
	parseDebounce = true
	task.defer(function()
		parseDebounce = false
		local text = CodeInput.Text
		local pos = CodeInput.CursorPosition
		local left = text:sub(1, pos - 1)

		local contextVar, token = left:match("([%w_]+)[%.:]([%w_]*)$")
		if not contextVar then token = left:match("([%w_]+)$") end

		if token and #token >= 1 then
			activeSuggestions = getSuggestions(token, contextVar)
			if #activeSuggestions > 0 then
				selectedIndex = 1
				local line, col = getCursorDetails()
				AutoPopup.Position = UDim2.fromOffset(MainFrame.AbsolutePosition.X + (col * 7.5) + 180, MainFrame.AbsolutePosition.Y + (line * 14.2) + 60)
				renderSuggestions()
				return
			end
		end
		AutoPopup.Visible = false
	end)
end

			renderSuggestions()

local function switchTab(tabId)
	activeTabId = tabId
	for _, t in ipairs(tabsData) do
		if t.frame then
			local isActive = (t.id == tabId)
			t.frame.BackgroundColor3 = isActive and Color3.fromRGB(24, 24, 24) or Color3.fromRGB(32, 32, 32)
			local indicator = t.frame:FindFirstChild("Indicator")
			if indicator then indicator.Visible = isActive end
		end
	end
	local activeT = getActiveTab()
	if activeT then
		CodeInput.Text = activeT.content or ""
		if activeT.filePath then
			BreadcrumbLabel.Text = formatBreadcrumb(activeT.filePath)
			FileNameInput.Text = getBasename(activeT.filePath)
		else
			BreadcrumbLabel.Text = "NarsEC/" .. activeT.filename
			FileNameInput.Text = activeT.filename
		end
		updateEditorState()
	end
end

local function closeTab(tabId)
	if #tabsData <= 1 then return end
	local targetIndex = nil
	for i, t in ipairs(tabsData) do
		if t.id == tabId then
			targetIndex = i
			if t.frame then t.frame:Destroy() end
			break
		end
	end
	if targetIndex then
		table.remove(tabsData, targetIndex)
		if activeTabId == tabId then
			local newActive = tabsData[math.min(targetIndex, #tabsData)]
			switchTab(newActive.id)
		else
			renderTabsUI()
		end
	end
end

local function renderDirectoryContents(dirPath, indentLevel, filterText)
	if not listfiles then return end
	local ok, items = pcall(function() return listfiles(dirPath) end)
	if not ok or not items then return end

	local folders = {}
	local files = {}

	for _, itemPath in ipairs(items) do
		local normPath = normalizePath(itemPath)
		local itemName = getBasename(normPath)

		local isDir = false
		if isfolder then
			pcall(function() isDir = isfolder(normPath) end)
		end

		if isDir then
			table.insert(folders, { path = normPath, name = itemName })
		else
			local isF = false
			if isfile then
				pcall(function() isF = isfile(normPath) end)
			end
			if isF or (not isfolder and not isfile) then
				local ext = itemName:match("%.([^%.]+)$")
				if ext then
					ext = ext:lower()
					if ext == "lua" or ext == "txt" or ext == "json" or ext == "md" then
						table.insert(files, { path = normPath, name = itemName })
					end
				end
			end
		end
	end

	table.sort(folders, function(a, b) return a.name:lower() < b.name:lower() end)
	table.sort(files, function(a, b) return a.name:lower() < b.name:lower() end)

	for _, fData in ipairs(folders) do
		local fPath = fData.path
		local fName = fData.name
		local isExpanded = folderExpandStates[fPath] or false
		local isFolderSelected = (selectedExplorerPath == fPath)

		local folderHeader = Instance.new("Frame")
		folderHeader.Size = UDim2.new(1, 0, 0, 20)
		folderHeader.BackgroundColor3 = isFolderSelected and Color3.fromRGB(0, 120, 215) or Color3.fromRGB(24, 24, 24)
		folderHeader.BackgroundTransparency = isFolderSelected and 0 or 1
		folderHeader.BorderSizePixel = 0
		folderHeader.Parent = ExpFileScroll

		local padLeft = 4 + (indentLevel * 12)

		local arrowIcon = Instance.new("ImageLabel")
		arrowIcon.Size = UDim2.fromOffset(10, 10)
		arrowIcon.Position = UDim2.new(0, padLeft, 0, 5)
		arrowIcon.BackgroundTransparency = 1
		arrowIcon.Image = isExpanded and ASSETS.ArrowDown or ASSETS.ArrowRight
		arrowIcon.ImageColor3 = isFolderSelected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(160, 160, 160)
		arrowIcon.Parent = folderHeader

		local folderIcon = Instance.new("ImageLabel")
		folderIcon.Size = UDim2.fromOffset(12, 12)
		folderIcon.Position = UDim2.new(0, padLeft + 14, 0, 4)
		folderIcon.BackgroundTransparency = 1
		folderIcon.Image = ASSETS.FolderIcon
		folderIcon.ImageColor3 = isFolderSelected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(220, 180, 100)
		folderIcon.Parent = folderHeader

		local folderLabel = Instance.new("TextLabel")
		folderLabel.Size = UDim2.new(1, -(padLeft + 48), 1, 0)
		folderLabel.Position = UDim2.new(0, padLeft + 30, 0, 0)
		folderLabel.BackgroundTransparency = 1
		folderLabel.Text = fName
		folderLabel.TextColor3 = isFolderSelected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(200, 200, 200)
		folderLabel.Font = Enum.Font.SourceSansBold
		folderLabel.TextSize = 11
		folderLabel.TextXAlignment = Enum.TextXAlignment.Left
		folderLabel.Parent = folderHeader

		local folderItemBtn = Instance.new("TextButton")
		folderItemBtn.Size = UDim2.new(1, -18, 1, 0)
		folderItemBtn.Position = UDim2.new(0, 0, 0, 0)
		folderItemBtn.BackgroundTransparency = 1
		folderItemBtn.Text = ""
		folderItemBtn.Parent = folderHeader

		local folderInsertBtn = Instance.new("ImageButton")
		folderInsertBtn.Size = UDim2.fromOffset(12, 12)
		folderInsertBtn.Position = UDim2.new(1, -16, 0, 4)
		folderInsertBtn.BackgroundTransparency = 1
		folderInsertBtn.Image = ASSETS.TabAdd
		folderInsertBtn.ImageColor3 = Color3.fromRGB(160, 160, 160)
		folderInsertBtn.Parent = folderHeader

		folderItemBtn.MouseButton1Click:Connect(function()
			selectedExplorerPath = fPath
			folderExpandStates[fPath] = not folderExpandStates[fPath]
			populateExplorer(ExpSearchInput and ExpSearchInput.Text or "")
		end)

		folderInsertBtn.MouseButton1Click:Connect(function()
			openInsertCard(fPath)
		end)

		if isExpanded then
			renderDirectoryContents(fPath, indentLevel + 1, filterText)
		end
	end

	for _, fileData in ipairs(files) do
		local itemPath = fileData.path
		local fileName = fileData.name

		if filterText == "" or fileName:lower():find(filterText, 1, true) then
			local isFileSelected = (selectedExplorerPath == itemPath)

			local itemFrame = Instance.new("Frame")
			itemFrame.Size = UDim2.new(1, 0, 0, 20)
			itemFrame.BackgroundColor3 = isFileSelected and Color3.fromRGB(0, 120, 215) or Color3.fromRGB(20, 20, 20)
			itemFrame.BackgroundTransparency = isFileSelected and 0 or 1
			itemFrame.BorderSizePixel = 0
			itemFrame.Parent = ExpFileScroll

			local padLeft = 4 + (indentLevel * 12) + 14

			local fileIcon = Instance.new("ImageLabel")
			fileIcon.Size = UDim2.fromOffset(12, 12)
			fileIcon.Position = UDim2.new(0, padLeft, 0, 4)
			fileIcon.BackgroundTransparency = 1
			fileIcon.Image = ASSETS.FileIcon
			fileIcon.ImageColor3 = isFileSelected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(160, 160, 160)
			fileIcon.Parent = itemFrame

			local fileLabel = Instance.new("TextLabel")
			fileLabel.Size = UDim2.new(1, -(padLeft + 34), 1, 0)
			fileLabel.Position = UDim2.new(0, padLeft + 16, 0, 0)
			fileLabel.BackgroundTransparency = 1
			fileLabel.Text = fileName
			fileLabel.TextColor3 = isFileSelected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 190, 190)
			fileLabel.Font = Enum.Font.SourceSans
			fileLabel.TextSize = 11
			fileLabel.TextXAlignment = Enum.TextXAlignment.Left
			fileLabel.Parent = itemFrame

			local fileItemBtn = Instance.new("TextButton")
			fileItemBtn.Size = UDim2.new(1, -18, 1, 0)
			fileItemBtn.Position = UDim2.new(0, 0, 0, 0)
			fileItemBtn.BackgroundTransparency = 1
			fileItemBtn.Text = ""
			fileItemBtn.Parent = itemFrame

			local fileInsertBtn = Instance.new("ImageButton")
			fileInsertBtn.Size = UDim2.fromOffset(12, 12)
			fileInsertBtn.Position = UDim2.new(1, -16, 0, 4)
			fileInsertBtn.BackgroundTransparency = 1
			fileInsertBtn.Image = ASSETS.TabAdd
			fileInsertBtn.ImageColor3 = Color3.fromRGB(160, 160, 160)
			fileInsertBtn.Parent = itemFrame

			fileItemBtn.MouseButton1Click:Connect(function()
				selectedExplorerPath = itemPath
				populateExplorer(ExpSearchInput and ExpSearchInput.Text or "")

				if readfile then
					local readSuccess, content = pcall(function() return readfile(itemPath) end)
					if readSuccess then
						local existingTab = nil
						for _, t in ipairs(tabsData) do
							if t.filePath == itemPath then
								existingTab = t
								break
							end
						end
						if existingTab then
							switchTab(existingTab.id)
						else
							addTab(fileName, content, "NarsEC", itemPath)
						end
						consoleSuccess("Loaded file: " .. formatBreadcrumb(itemPath))
					else
						consoleError("Failed to read file: " .. fileName)
					end
				else
					consoleWarning("readfile API unavailable")
				end
			end)

			fileInsertBtn.MouseButton1Click:Connect(function()
				local parentDir = getDirectory(itemPath)
				openInsertCard(parentDir)
			end)
		end
	end
end

function populateExplorer(filterText)
	for _, child in ipairs(ExpFileScroll:GetChildren()) do
		if child:IsA("TextButton") or child:IsA("Frame") then 
			child:Destroy() 
		end
	end

	filterText = (filterText or ""):lower()

	if isfolder then
		local okRoot, hasRoot = pcall(function() return isfolder(ROOT_PATH) end)
		if okRoot and hasRoot then
			renderDirectoryContents(ROOT_PATH, 0, filterText)
		end
	end

	ExpFileScroll.CanvasSize = UDim2.new(0, 0, 0, ExpFileLayout.AbsoluteContentSize.Y)
end

DeleteFileBtn.MouseButton1Click:Connect(function()
	local curTab = getActiveTab()
	if not curTab then
		consoleWarning("No active tab to delete")
		return
	end

	local filePath = curTab.filePath
	if not filePath or filePath == "" then
		consoleWarning("Active tab is not saved on disk")
		return
	end

	if isfile and delfile then
		if isfile(filePath) then
			local success, err = pcall(function()
				delfile(filePath)
			end)
			if success then
				curTab.filePath = nil
				consoleSuccess("File deleted successfully: " .. formatBreadcrumb(filePath))
				populateExplorer(ExpSearchInput.Text)
			else
				consoleError("Failed to delete file: " .. tostring(err))
			end
		else
			consoleWarning("File does not exist: " .. filePath)
		end
	else
		consoleWarning("FileSystem API unavailable")
	end
end)
